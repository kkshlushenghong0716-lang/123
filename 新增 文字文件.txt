<script>
/** 難度與全局配置 **/
const DIFF_SETTINGS = {
    easy:   { money: 850, hp: 20, growth: 0.15, speed: 0.95 },
    normal: { money: 600, hp: 10, growth: 0.30, speed: 1.05 },
    hell:   { money: 400, hp: 3,  growth: 0.50, speed: 1.25 }
};

const CONFIG = {
    grid: 40,
    path: [], 
    towers: {
        basic:  { cost: 100, d: 50,  cd: 22, col: '#00f2ff', name: '等離子砲' },
        sniper: { cost: 350, d: 450, cd: 120, col: '#bc13fe', name: '磁軌砲' },
        slow:   { cost: 180, d: 20,  cd: 45, col: '#f1c40f', name: 'EMP脈衝' },
        laser:  { cost: 260, d: 5,   cd: 1,  col: '#ff2a85', name: '裂變雷射' } 
    }
};

const canvas = document.getElementById('c');
const ctx = canvas.getContext('2d', { alpha: false }); 

const bgCanvas = document.createElement('canvas');
bgCanvas.width = 800; bgCanvas.height = 600;
const bgCtx = bgCanvas.getContext('2d', { alpha: false });

const uiMoney = document.getElementById('m-val');
const uiHealth = document.getElementById('h-val');
const uiWave = document.getElementById('w-val');
let lastUI = { m: -1, h: -1, w: -1 };

let money, health, wave = 1, frame = 0, isRunning = false, difficulty = 'normal';
let enemies = [], towers = [], bullets = [], particles = [], fxRing = [];
let mouse = {x:0, y:0}, selType = null, selCost = 0, selRange = 0, selectedTower = null, shake = 0;
let gameSpeed = 1; 

function generateRandomPath() {
    let p = [];
    p.push({x: -40, y: 120});
    p.push({x: 240 + Math.random() * 80, y: 120});
    let midY = 280 + Math.random() * 80;
    p.push({x: p[1].x, y: midY});
    let turnX = 480 + Math.random() * 80;
    p.push({x: turnX, y: midY});
    p.push({x: turnX, y: 480});
    p.push({x: 840, y: 480});
    CONFIG.path = p;
}

function preRenderBackground() {
    bgCtx.fillStyle = '#030508';
    bgCtx.fillRect(0, 0, 800, 600);
    
    bgCtx.strokeStyle = '#05090e';
    bgCtx.lineWidth = 1;
    bgCtx.beginPath();
    for(let x=0; x<800; x+=CONFIG.grid) { bgCtx.moveTo(x, 0); bgCtx.lineTo(x, 600); }
    for(let y=0; y<600; y+=CONFIG.grid) { bgCtx.moveTo(0, y); bgCtx.lineTo(800, y); }
    bgCtx.stroke();

    bgCtx.strokeStyle = '#0b131f'; bgCtx.lineWidth = 36; bgCtx.lineCap = 'round'; bgCtx.lineJoin = 'round';
    bgCtx.beginPath(); 
    CONFIG.path.forEach((p,i) => i===0 ? bgCtx.moveTo(p.x, p.y) : bgCtx.lineTo(p.x, p.y)); 
    bgCtx.stroke();
    
    bgCtx.strokeStyle = '#112235'; bgCtx.lineWidth = 4; bgCtx.stroke();
}

function toggleSpeed() {
    gameSpeed = gameSpeed === 1 ? 2 : (gameSpeed === 2 ? 3 : 1);
    document.getElementById('speed-btn').innerText = `速度: ${gameSpeed}X`;
}

function distSq(x1, y1, x2, y2) {
    let dx = x1 - x2;
    let dy = y1 - y2;
    return dx * dx + dy * dy;
}

function isPointNearPath(x, y) {
    for (let i = 0; i < CONFIG.path.length - 1; i++) {
        let p1 = CONFIG.path[i], p2 = CONFIG.path[i+1];
        let l2 = distSq(p1.x, p1.y, p2.x, p2.y);
        let t = ((x - p1.x) * (p2.x - p1.x) + (y - p1.y) * (p2.y - p1.y)) / l2;
        t = Math.max(0, Math.min(1, t));
        let projX = p1.x + t * (p2.x - p1.x);
        let projY = p1.y + t * (p2.y - p1.y);
        if (distSq(x, y, projX, projY) < 784) return true; 
    }
    return false;
}

class Particle {
    constructor(x, y, col) {
        this.x = x; this.y = y; this.col = col;
        this.vx = (Math.random()-0.5)*8; this.vy = (Math.random()-0.5)*8;
        this.life = 1.0; this.size = Math.random()*3 + 1.5;
    }
    update() {
        this.x += this.vx * gameSpeed; 
        this.y += this.vy * gameSpeed; 
        this.vx *= 0.95; 
        this.vy *= 0.95;
        this.life -= 0.05 * gameSpeed;
    }
    draw(ctx) {
        ctx.fillStyle = this.col; ctx.globalAlpha = Math.max(0, this.life); 
        ctx.fillRect(this.x, this.y, this.size, this.size); 
    }
}

class Enemy {
    constructor(type = 'scout') {
        this.id = Math.random();
        this.x = CONFIG.path[0].x; this.y = CONFIG.path[0].y; this.pIdx = 0;
        this.type = type;
        const set = DIFF_SETTINGS[difficulty];
        const hpFactor = 1 + (wave - 1) * set.growth;
        
        const stats = {
            scout:  { h: 140, s: 3.5, r: 8, col: '#34d399', sh: 0, rew: 18 },
            ghost:  { h: 90,  s: 5.2, r: 6, col: '#ffffff', sh: 0, rew: 22 }, 
            tank:   { h: 1600, s: 0.9, r: 16, col: '#f87171', sh: 0, rew: 65 }, 
            shield: { h: 600, s: 2.0, r: 13, col: '#60a5fa', sh: 700, rew: 55 }, 
            boss:   { h: 12000, s: 0.65, r: 28, col: '#bc13fe', sh: 4000, rew: 300 } 
        };

        const s = stats[type];
        this.h = s.h * hpFactor; this.maxH = this.h;
        this.sh = s.sh * hpFactor; this.maxSh = this.sh;
        this.s = s.s * set.speed; this.r = s.r; this.col = s.col;
        this.rew = s.rew; 
        this.angle = 0; this.hitT = 0; this.slow = 0; this.progress = 0;
    }
    update() {
        if (this.hitT > 0) this.hitT -= gameSpeed;
        
        let finalSlow = (this.slow > 0 && this.type !== 'ghost') ? (this.slow -= gameSpeed, this.s * 0.50) : this.s;
        let curS = finalSlow * gameSpeed; 
        let t = CONFIG.path[this.pIdx + 1];
        if (t) {
            let dx = t.x - this.x, dy = t.y - this.y;
            let dSq = dx*dx + dy*dy;
            this.angle = Math.atan2(dy, dx);
            if (dSq < curS * curS) {
                this.pIdx++;
            } else { 
                let d = Math.sqrt(dSq);
                this.x += (dx / d) * curS; 
                this.y += (dy / d) * curS; 
                this.progress += curS; 
            }
        }
        return (this.x > 820);
    }
    draw(ctx) {
        ctx.save(); ctx.translate(this.x, this.y);
        if(this.type === 'ghost') ctx.globalAlpha = 0.4;
        
        if(this.sh > 0) {
            ctx.beginPath(); ctx.arc(0, 0, this.r+7, 0, Math.PI*2); 
            ctx.strokeStyle = 'rgba(96, 165, 250, 0.8)'; ctx.lineWidth = 2; ctx.stroke();
        }
        
        ctx.rotate(this.angle); 
        ctx.fillStyle = this.hitT > 0 ? '#fff' : this.col;
        ctx.beginPath();
        if(this.type === 'scout' || this.type === 'ghost') { 
            ctx.moveTo(this.r*1.3, 0); ctx.lineTo(-this.r, -this.r*0.8); ctx.lineTo(-this.r, this.r*0.8); 
        } else if(this.type === 'tank' || this.type === 'shield') { 
            ctx.rect(-this.r, -this.r, this.r*2, this.r*2); 
        } else { 
            for(let i=0; i<8; i++) { 
                let a = Math.PI/4*i; let r2 = i%2?this.r:this.r*0.6; 
                ctx.lineTo(Math.cos(a)*r2, Math.sin(a)*r2); 
            } 
        }
        ctx.fill(); ctx.restore();
        
        const bw = this.r * 2.5;
        ctx.fillStyle = '#1e293b'; ctx.fillRect(this.x-bw/2, this.y-this.r-15, bw, 4);
        ctx.fillStyle = this.col; ctx.fillRect(this.x-bw/2, this.y-this.r-15, bw*(Math.max(0, this.h)/this.maxH), 4);
        if(this.maxSh > 0 && this.sh > 0) {
            ctx.fillStyle = '#1e293b'; ctx.fillRect(this.x-bw/2, this.y-this.r-9, bw, 2);
            ctx.fillStyle = '#60a5fa'; ctx.fillRect(this.x-bw/2, this.y-this.r-9, bw*(this.sh/this.maxSh), 2);
        }
    }
}

function initGame(mode) {
    generateRandomPath();
    preRenderBackground();
    difficulty = mode;
    const set = DIFF_SETTINGS[mode];
    money = set.money; health = set.hp; wave = 1; frame = 0;
    enemies = []; towers = []; bullets = []; particles = []; fxRing = [];
    
    document.getElementById('start-screen').style.display = 'none';
    document.getElementById('hud').style.opacity = 1;
    document.getElementById('shop').style.transform = 'translateY(0)';
    isRunning = true;
    animate();
}

function animate() {
    if(!isRunning) return;
    frame += gameSpeed; 
    let intFrame = Math.floor(frame);
    
    ctx.save();
    if(shake > 0) { ctx.translate((Math.random()-0.5)*shake, (Math.random()-0.5)*shake); shake *= 0.88; }
    
    ctx.drawImage(bgCanvas, 0, 0);

    // 處理敵人生成
    if (wave <= 10) {
        let spawnInterval = Math.max(12, 75 - wave*4);
        if(intFrame % spawnInterval < gameSpeed && intFrame - Math.floor(frame - gameSpeed) >= 1) {
            let r = Math.random(), type = 'scout';
            if(wave >= 2 && r > 0.78) type = 'ghost';
            if(wave >= 4 && r > 0.52 && r <= 0.78) type = 'tank';
            if(wave >= 6 && r > 0.32 && r <= 0.52) type = 'shield';
            if(wave === 10 || (wave % 5 === 0 && enemies.filter(e=>e.type==='boss').length === 0 && Math.random() < 0.15)) type = 'boss';
            enemies.push(new Enemy(type));
        }
    }
    
    // 處理波次推進
    if(wave < 10 && intFrame % 1300 < gameSpeed && intFrame - Math.floor(frame - gameSpeed) >= 1) {
        wave++;
    }

    // 【新增修改】被動自動加錢機制：每 60 幀（約 1 秒，受 2X、3X 加速影響）自動增加 10 元
    if (intFrame % 60 < gameSpeed && intFrame - Math.floor(frame - gameSpeed) >= 1) {
        money += 10;
    }

    // 處理防禦塔與攻擊
    towers.forEach(t => {
        if(t.cd > 0) t.cd -= gameSpeed; 
        let rSq = t.r * t.r;
        let targets = enemies.filter(e => distSq(e.x, e.y, t.x, t.y) < rSq);
        
        if(targets.length > 0) {
            let target = targets.sort((a,b) => b.progress - a.progress)[0];
            t.rot = Math.atan2(target.y - t.y, target.x - t.x);
            
            if(t.type === 'laser') {
                target.hitT = 2;
                let finalDmg = (target.type === 'tank' || target.type === 'boss') ? t.d * 1.4 : t.d; 
                
                if(target.sh > 0) {
                    target.sh -= finalDmg * gameSpeed;
                    if(target.sh < 0) { target.h += target.sh; target.sh = 0; }
                } else { target.h -= finalDmg * gameSpeed; }
                
                ctx.beginPath();
                ctx.strokeStyle = t.col; ctx.lineWidth = 1.5 + Math.random()*2 + t.lvl;
                ctx.moveTo(t.x, t.y); ctx.lineTo(target.x, target.y); ctx.stroke();
            } else if(t.cd <= 0) {
                bullets.push({ 
                    x: t.x, y: t.y, t: target, tId: target.id, 
                    d: t.d, col: t.col, type: t.type, s: t.type==='sniper'? 24:16, trail: []
                });
                t.cd = t.maxCd;
            }
        }
        t.draw(ctx);
    });

    // 處理子彈
    bullets = bullets.filter(b => {
        let targetExist = enemies.find(e => e.id === b.tId);
        if(!targetExist) {
            for(let i=0; i<3; i++) particles.push(new Particle(b.x, b.y, b.col));
            return false;
        }
        let dx = b.t.x - b.x, dy = b.t.y - b.y;
        let dSq = dx*dx + dy*dy;
        b.trail.push({x: b.x, y: b.y}); if(b.trail.length > 5) b.trail.shift();
        
        let step = b.s * gameSpeed; 
        if(dSq < (step + 12) * (step + 12)) {
            b.t.hitT = 3;
            if(b.t.sh > 0) { 
                b.t.sh -= b.d; if(b.t.sh < 0) { b.t.h += b.t.sh; b.t.sh = 0; } 
            } else { b.t.h -= b.d; }
            
            if(b.type === 'slow') {
                b.t.slow = 90;
                fxRing.push({x: b.x, y: b.y, r: 10, maxR: 50, col: b.col, alpha: 1});
            }
            if(b.type === 'sniper') {
                shake = 10;
                fxRing.push({x: b.x, y: b.y, r: 5, maxR: 35, col: '#fff', alpha: 1});
            }
            
            let pCount = b.type === 'sniper' ? 12 : 6;
            for(let i=0; i<pCount; i++) particles.push(new Particle(b.x, b.y, b.col));
            return false;
        }
        let dist = Math.sqrt(dSq);
        b.x += (dx/dist)*step; b.y += (dy/dist)*step;
        
        ctx.beginPath(); ctx.strokeStyle = b.col; ctx.lineWidth = b.type==='sniper'? 4 : 2;
        if(b.trail.length > 1) {
            b.trail.forEach((p, i) => i===0?ctx.moveTo(p.x,p.y):ctx.lineTo(p.x,p.y)); ctx.stroke();
        }
        return true;
    });

    // 處理特效環
    ctx.lineWidth = 2;
    fxRing = fxRing.filter(r => {
        r.r += 3 * gameSpeed; r.alpha -= 0.05 * gameSpeed;
        ctx.beginPath(); ctx.arc(r.x, r.y, r.r, 0, Math.PI*2);
        ctx.strokeStyle = r.col; ctx.globalAlpha = Math.max(0, r.alpha); ctx.stroke();
        return r.alpha > 0;
    });
    ctx.globalAlpha = 1;

    // 處理敵人
    enemies = enemies.filter(e => {
        if(e.update()) { health--; shake = 15; return false; } 
        if(e.h <= 0) { 
            money += e.rew; 
            for(let i=0; i<15; i++) particles.push(new Particle(e.x, e.y, e.col)); 
            return false; 
        }
        e.draw(ctx); return true;
    });

    // 處理粒子
    particles.forEach(p => p.update());
    particles = particles.filter(p => { p.draw(ctx); return p.life > 0; });
    ctx.globalAlpha = 1;
    
    // 繪製滑鼠預覽框
    if(selType && !selectedTower) {
        let currentGridX = Math.floor(mouse.x / 40), currentGridY = Math.floor(mouse.y / 40);
        let centerX = currentGridX * 40 + 20, centerY = currentGridY * 40 + 20;
        let isValid = !isPointNearPath(centerX, centerY) && !towers.find(t => t.gx === currentGridX && t.gy === currentGridY) && centerX < 800 && centerY < 600;

        ctx.beginPath(); ctx.arc(centerX, centerY, selRange, 0, Math.PI*2);
        ctx.fillStyle = isValid ? 'rgba(0, 242, 255, 0.04)' : 'rgba(248, 113, 113, 0.06)'; ctx.fill();

        let boxColor = isValid ? 'rgba(0, 242, 255, ' : 'rgba(248, 113, 113, ';
        if(currentGridX >= 0 && currentGridX < 20 && currentGridY >= 0 && currentGridY < 15) {
            ctx.fillStyle = boxColor + '0.35)'; ctx.strokeStyle = boxColor + '0.7)'; ctx.lineWidth = 2;
            ctx.fillRect(currentGridX * 40 + 2, currentGridY * 40 + 2, 36, 36);
            ctx.strokeRect(currentGridX * 40 + 2, currentGridY * 40 + 2, 36, 36);
        }
    }
    ctx.restore();

    // UI 快取更新
    let curM = Math.floor(money);
    if(curM !== lastUI.m) { uiMoney.innerText = curM; lastUI.m = curM; }
    if(health !== lastUI.h) { uiHealth.innerText = health; lastUI.h = health; }
    if(wave !== lastUI.w) { uiWave.innerText = wave; lastUI.w = wave; }

    // 檢查結束條件
    if(health <= 0) { 
        isRunning = false; 
        document.getElementById('game-over').style.display = 'flex'; 
        document.getElementById('final-wave').innerText = `系統阻斷波次: ${wave}/10 (${difficulty.toUpperCase()})`; 
    } else if(wave === 10 && enemies.length === 0) {
        isRunning = false;
        document.getElementById('game-victory').style.display = 'flex';
        document.getElementById('victory-detail').innerText = `挑戰模式：${difficulty.toUpperCase()} │ 剩餘核心安全係數：${health}`;
    } else {
        requestAnimationFrame(animate);
    }
}

class TowerInstance {
    constructor(gx, gy, type) {
        this.gx = gx; this.gy = gy; this.x = gx*40+20; this.y = gy*40+20;
        this.type = type; this.lvl = 1; this.rot = 0; this.cd = 0; this.sync();
    }
    sync() {
        let b = CONFIG.towers[this.type];
        this.d = b.d * (1 + (this.lvl-1)*1.4);
        this.r = (this.type==='sniper'? 360 : (this.type==='slow'? 140 : (this.type==='laser'? 170 : 130))) * (1 + (this.lvl-1)*0.15);
        this.maxCd = b.cd * (this.lvl === 3 ? 0.75 : 1);
        this.col = b.col; this.name = b.name;
    }
    draw(ctx) {
        ctx.save(); ctx.translate(this.x, this.y);
        ctx.strokeStyle = (selectedTower === this) ? 'var(--c)' : 'rgba(255,255,255,0.15)';
        ctx.lineWidth = (selectedTower === this) ? 2 : 1;
        ctx.strokeRect(-15,-15,30,30);
        
        ctx.fillStyle = this.col;
        for(let i=0; i<this.lvl; i++) { 
            ctx.beginPath(); ctx.arc(-10+i*10, -20, 2.5, 0, Math.PI*2); ctx.fill(); 
        }
        
        ctx.rotate(this.rot); 
        ctx.fillStyle = '#0f172a'; ctx.fillRect(-4, -6, 22, 12);
        ctx.strokeStyle = this.col; ctx.lineWidth = 2; ctx.strokeRect(-4, -6, 22, 12);
        
        if(this.type === 'sniper' && this.cd > 0) {
            ctx.fillStyle = '#fff'; ctx.fillRect(4, -2, 4, 4);
        }
        ctx.restore();
    }
}

canvas.onclick = (e) => {
    const rect = canvas.getBoundingClientRect();
    const x = (e.clientX - rect.left) * (800 / rect.width), y = (e.clientY - rect.top) * (600 / rect.height);
    let gx = Math.floor(x/40), gy = Math.floor(y/40);
    if(gx < 0 || gx >= 20 || gy < 0 || gy >= 15) return;
    
    let t = towers.find(t => t.gx === gx && t.gy === gy);
    if(t) { showUpgrade(t); return; }
    
    if(selType) {
        if(isPointNearPath(gx*40+20, gy*40+20)) return;
        if(money < selCost) { 
            let msg = document.getElementById('msg'); msg.style.opacity = 1; 
            setTimeout(()=>msg.style.opacity=0, 1000); return; 
        }
        towers.push(new TowerInstance(gx, gy, selType)); 
        money -= selCost; selType = null;
        document.querySelectorAll('.item').forEach(i => i.classList.remove('active'));
    } else { closeMenu(); }
};

function showUpgrade(t) {
    selectedTower = t; selType = null;
    document.querySelectorAll('.item').forEach(i => i.classList.remove('active'));
    document.getElementById('upgrade-panel').style.display = 'flex';
    document.getElementById('unit-name').innerText = `${t.name} 【階級 ${t.lvl}】`;
    const upBtn = document.getElementById('up-btn');
    
    let baseCost = CONFIG.towers[t.type].cost;
    if(t.lvl < 3) {
        let upgradeCost = Math.floor(baseCost * (t.lvl === 1 ? 1.2 : 1.8));
        upBtn.innerText = `吞吐量強化 $${upgradeCost}`;
        upBtn.style.background = 'var(--c)';
        upBtn.onclick = () => { 
            if(money >= upgradeCost) { money -= upgradeCost; t.lvl++; t.sync(); showUpgrade(t); } 
            else {
                let msg = document.getElementById('msg'); msg.innerText = "能量儲備不足"; msg.style.opacity = 1; 
                setTimeout(()=>msg.style.opacity=0, 1000);
            }
        };
    } else { 
        upBtn.innerText = "核心已達極限"; upBtn.style.background = '#334155'; upBtn.onclick = null; 
    }
    document.getElementById('sell-btn').onclick = () => { 
        money += Math.floor(baseCost * 0.6 * t.lvl); 
        towers = towers.filter(tw => tw !== t); closeMenu(); 
    };
}

function closeMenu() { selectedTower = null; document.getElementById('upgrade-panel').style.display = 'none'; }

function sel(type, cost, range, el) { 
    selType = type; selCost = cost; selRange = range; selectedTower = null; 
    document.querySelectorAll('.item').forEach(i => i.classList.remove('active')); 
    el.classList.add('active'); closeMenu(); 
}
